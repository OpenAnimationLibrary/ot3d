cmake_minimum_required(VERSION 3.14)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(OT3DExrDocument LANGUAGES C CXX)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/OpenEXR.cmake")
    set(EXR_DOCUMENT_STANDALONE ON)
else()
    set(EXR_DOCUMENT_STANDALONE OFF)
endif()

add_library(otexrdocument STATIC deepimage.cpp deepimage.h exrdocument.cpp exrdocument.h)
target_compile_features(otexrdocument PUBLIC cxx_std_17)
target_include_directories(otexrdocument PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(otexrdocument PRIVATE OpenEXR::OpenEXR)
set_target_properties(otexrdocument PROPERTIES POSITION_INDEPENDENT_CODE ON)

option(BUILD_EXR_DOCUMENT_TESTS "Build EXR document and deep transport checks" ${EXR_DOCUMENT_STANDALONE})
if(BUILD_EXR_DOCUMENT_TESTS)
    enable_testing()
    add_executable(exr_inspect exr_inspect.cpp)
    target_link_libraries(exr_inspect PRIVATE otexrdocument)
    add_executable(exrdocument_test tests/exrdocument_test.cpp)
    target_link_libraries(exrdocument_test PRIVATE otexrdocument OpenEXR::OpenEXR)
    add_test(NAME exrdocument COMMAND exrdocument_test)
endif()
