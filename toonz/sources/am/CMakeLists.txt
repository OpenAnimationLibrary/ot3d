cmake_minimum_required(VERSION 3.10)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(OT3DAnimationMaster LANGUAGES CXX)
    set(AM_STANDALONE ON)
else()
    set(AM_STANDALONE OFF)
endif()
add_library(otam STATIC
    ammodel.cpp ammodel.h
    ammaterial.cpp ammaterial.h
    amaction.cpp amaction.h
    amink.cpp amink.h
)
target_compile_features(otam PUBLIC cxx_std_17)
target_include_directories(otam PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(otam PROPERTIES POSITION_INDEPENDENT_CODE ON)
option(BUILD_AM_TESTS "Build native A:M parser/spatial regression tests" ${AM_STANDALONE})
if(BUILD_AM_TESTS)
    enable_testing()
    add_executable(ammodel_test tests/ammodel_test.cpp)
    target_link_libraries(ammodel_test PRIVATE otam)
    add_test(NAME ammodel COMMAND ammodel_test)
    add_executable(ammaterial_test tests/ammaterial_test.cpp)
    target_link_libraries(ammaterial_test PRIVATE otam)
    add_test(NAME ammaterial COMMAND ammaterial_test)
    add_executable(amaction_test tests/amaction_test.cpp)
    target_link_libraries(amaction_test PRIVATE otam)
    add_test(NAME amaction COMMAND amaction_test)
    add_executable(am_inspect am_inspect.cpp)
    target_link_libraries(am_inspect PRIVATE otam)
endif()

if(BUILD_AM_LEVEL_TESTS AND NOT AM_STANDALONE)
    add_executable(amlevel_test tests/amlevel_test.cpp)
    target_link_libraries(amlevel_test PRIVATE
        toonzlib toonzqt Qt5::Core Qt5::Gui Qt5::Widgets
    )
    add_test(NAME amlevel COMMAND amlevel_test)
    ot_configure_native_test_runtime(amlevel)
    set_property(TEST amlevel APPEND PROPERTY ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()

if(BUILD_AM_LEVEL_TESTS AND NOT AM_STANDALONE)
    add_executable(amfx_test tests/amfx_test.cpp)
    # The test compiles the FX implementations directly into its executable.
    target_compile_definitions(amfx_test PRIVATE TNZSTDFX_EXPORTS)
    target_link_libraries(amfx_test PRIVATE toonzlib toonzqt Qt5::Widgets)
    add_test(NAME amfx COMMAND amfx_test)
    ot_configure_native_test_runtime(amfx)
    set_property(TEST amfx APPEND PROPERTY ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()

if(BUILD_AM_LEVEL_TESTS AND NOT AM_STANDALONE)
    add_executable(amcamera_test tests/amcamera_test.cpp)

    target_link_libraries(amcamera_test PRIVATE toonzlib toonzqt Qt5::Widgets)
